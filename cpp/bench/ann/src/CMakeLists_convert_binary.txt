# Build the binary quantization conversion utility
# To be added to the main CMakeLists.txt if needed

add_executable(convert_binary_quantize convert_binary_quantize.cpp)

target_link_libraries(
  convert_binary_quantize
  PRIVATE
    cuvs::cuvs
    raft::raft
    CUDA::cudart
    $<$<BOOL:${CUDAToolkit_FOUND}>:CUDA::cublas>
    $<TARGET_NAME_IF_EXISTS:OpenMP::OpenMP_CXX>
    $<TARGET_NAME_IF_EXISTS:conda_env>
)

target_compile_options(
  convert_binary_quantize
  PRIVATE
    $<$<COMPILE_LANGUAGE:CXX>:${CUVS_CXX_FLAGS}>
    $<$<COMPILE_LANGUAGE:CUDA>:${CUVS_CUDA_FLAGS}>
)

install(TARGETS convert_binary_quantize DESTINATION bin)
